// ✅ الحل الصحيح
const ALLOWED_TYPES = ['driver', 'customer'] as const;
type DownloadType = typeof ALLOWED_TYPES[number];

export default function DownloadPage() {
  const searchParams = useSearchParams();
  const rawType = searchParams.get('type') || 'customer';

  const type: DownloadType = ALLOWED_TYPES.includes(rawType as DownloadType)
    ? (rawType as DownloadType)
    : 'customer';
}
